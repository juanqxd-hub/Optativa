#include <LiquidCrystal.h>

// Pines: RS, E, D4, D5, D6, D7
LiquidCrystal lcd(7, 6, 5, 4, 3, 2);

float suma;

void setup() {
  lcd.begin(16, 2);
}
void loop() {

  suma = 0;

  // Tomar 5 muestras
  for (int i = 0; i < 5; i++) {

    int sensor = analogRead(A0);

    float temperatura = (sensor * 5000.0 / 1023.0) / 10.0;

    suma = suma + temperatura;

    delay(500);
  }

  // Calcular promedio
  float promedio = suma / 5.0;

  // Mostrar temperatura
  lcd.setCursor(0, 0);
  lcd.print("Temp: ");
  lcd.print(promedio, 1);
  lcd.print(" C");
}
