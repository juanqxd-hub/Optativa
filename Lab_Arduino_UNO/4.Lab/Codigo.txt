#include <LiquidCrystal.h>
#include "DHT.h"

// ----------------------
// LCD
// RS, E, D4, D5, D6, D7
LiquidCrystal lcd(7, 6, 5, 4, 3, 2);

// ----------------------
// DHT22
#define SENSOR 8
#define TIPO_DHT DHT22

DHT dht(SENSOR, TIPO_DHT);

void setup() {

  // Monitor serial
  Serial.begin(9600);

  // Iniciar LCD
  lcd.begin(16, 2);

  // Iniciar DHT22
  dht.begin();

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Iniciando...");
  
  delay(2000);
}

void loop() {

  // Leer sensor
  float t = dht.readTemperature();
  float h = dht.readHumidity();

  // Comprobar error
  if (isnan(t) || isnan(h)) {

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Error DHT22");

    Serial.println("Error al leer DHT22");

    delay(2000);
    return;
  }

  // ----------------------
  // Monitor serial

  Serial.print("Temperatura: ");
  Serial.print(t);
  Serial.print(" C  Humedad: ");
  Serial.print(h);
  Serial.println(" %");

  // ----------------------
  // LCD

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("Temp: ");
  lcd.print(t, 1);
  lcd.print((char)223);
  lcd.print("C");

  lcd.setCursor(0, 1);
  lcd.print("Hum:  ");
  lcd.print(h, 1);
  lcd.print("%");

  delay(2000);
}
